import pandas as pd

def simple_backtest(signals_df, capital=1_000_000):
    if signals_df.empty:
        return pd.DataFrame()
    signals_df = signals_df.copy()
    signals_df['return'] = signals_df['prob'] * (signals_df['tp'] - signals_df['price']) / (signals_df['price'] - signals_df['sl']) - (1 - signals_df['prob'])
    signals_df['pnl'] = signals_df['return'] * capital * signals_df['position_size']
    signals_df['cum_pnl'] = signals_df['pnl'].cumsum()
    return signals_df

async def get_price_changes(client, figi):
    """
    Возвращает текущую цену и изменения за заданные таймфреймы
    """
    now = datetime.now(timezone.utc)
    changes = {}
    current_price = None

    # Получим данные за последний день (макс. интервал)
    from_time = now - timedelta(days=1)
    candles = await client.market_data.get_candles(
        figi=figi,
        from_=from_time,
        to=now,
        interval=1  # 1-минутные свечи
    )

    if not candles.candles:
        return None, None, {}

    # Текущая цена — последняя закрытая свеча
    current_price = quote_to_float(candles.candles[-1].close)

    # Создадим словарь: timestamp -> цена закрытия
    price_by_minute = {}
    for c in candles.candles:
        ts = c.time.replace(second=0, microsecond=0)
        price_by_minute[ts] = quote_to_float(c.close)

    # Сортируем временные метки
    timestamps = sorted(price_by_minute.keys())

    # Для каждого таймфрейма найдём цену N минут назад
    for label, minutes in TIMEFRAMES.items():
        target_time = now.replace(second=0, microsecond=0) - timedelta(minutes=minutes)
        # Найдём ближайшую цену <= target_time
        price_ago = None
        for ts in reversed(timestamps):
            if ts <= target_time:
                price_ago = price_by_minute[ts]
                break
        if price_ago:
            change = (current_price - price_ago) / price_ago * 100
            changes[label] = round(change, 2)
        else:
            changes[label] = 0.0

    return current_price, changes

# ml.py
import pandas as pd
import numpy as np
import lightgbm as lgb


def prepare_features(df):
    df = df.copy()
    df['return'] = df['close'].pct_change()
    df['volatility'] = df['return'].rolling(5).std()
    df['ma5'] = df['close'].rolling(5).mean()
    df['ma20'] = df['close'].rolling(20).mean()
    df['ma_ratio'] = df['ma5'] / df['ma20']
    df['volume_ma'] = df['volume'].rolling(5).mean()
    df['volume_ratio'] = df['volume'] / df['volume_ma']
    df['high_low'] = (df['high'] - df['low']) / df['close']
    df['next_return'] = df['return'].shift(-1)
    df['target'] = (df['next_return'] > 0).astype(int)
    return df.dropna()


def train_model(candles):
    df = pd.DataFrame([{
        'open': c.open.units + c.open.nano / 1e9,
        'high': c.high.units + c.high.nano / 1e9,
        'low': c.low.units + c.low.nano / 1e9,
        'close': c.close.units + c.close.nano / 1e9,
        'volume': c.volume
    } for c in candles])

    df = prepare_features(df)
    if len(df) < 10:
        raise ValueError("Недостаточно данных для обучения")

    features = ['return', 'volatility', 'ma_ratio', 'volume_ratio', 'high_low']
    X = df[features]
    y = df['target']

    model = lgb.LGBMClassifier(n_estimators=30, random_state=42, verbose=-1)
    model.fit(X, y)
    return model, features


def predict(model, features, X_row):
    prob = model.predict_proba([X_row])[0, 1]
    return prob

from config import (
    MAX_RISK_PER_TRADE,
    MAX_RISK_PER_ASSET,
    MAX_TOTAL_PORTFOLIO_RISK,
    MAX_DRAWDOWN,
    INITIAL_CAPITAL

from data_base import get_total_pnl, log_performance

class PortfolioRiskManager:
    def __init__(self):
        self.current_balance = INITIAL_CAPITAL
        self.active_risk = 0.0
        self.asset_risk = {}

    def update_balance(self):
        pnl = get_total_pnl()
        self.current_balance = INITIAL_CAPITAL + pnl
        drawdown = log_performance(self.current_balance)
        if drawdown > MAX_DRAWDOWN:
            raise Exception("🛑 Максимальная просадка превышена! Торговля остановлена.")

    def can_add_position(self, figi, position_risk):
        if position_risk > MAX_RISK_PER_TRADE * self.current_balance:
            return False
        current_asset_risk = self.asset_risk.get(figi, 0)
        if (current_asset_risk + position_risk) > MAX_RISK_PER_ASSET * self.current_balance:
            return False
        if (self.active_risk + position_risk) > MAX_TOTAL_PORTFOLIO_RISK * self.current_balance:
            return False
        return True

    def commit_risk(self, figi, risk):
        self.active_risk += risk
        self.asset_risk[figi] = self.asset_risk.get(figi, 0) + risk

    def reduce_risk(self, figi, risk):
        self.active_risk = max(0, self.active_risk - risk)
        self.asset_risk[figi] = max(0, self.asset_risk.get(figi, 0) - risk)

            to_time = datetime.now(timezone.utc)
            from_time = to_time - timedelta(minutes=60)
            candles = await client.market_data.get_candles(
                figi=figi,
                from_=from_time,
                to=to_time,
                interval=CandleInterval.CANDLE_INTERVAL_1_MIN
            )
            if len(candles.candles) < 10:
                continue

            model, features = train_model(candles.candles)
            last = candles.candles[-1]
            df = prepare_features(pd.DataFrame([{
                'open': quote_to_float(last.open),
                'high': quote_to_float(last.high),
                'low': quote_to_float(last.low),
                'close': quote_to_float(last.close),
                'volume': last.volume
            }]))
            if df.empty or len(df) == 0:
                continue

            X_row = df[features].iloc[-1].tolist()
            prob = predict(model, features, X_row)
            price = df['close'].iloc[-1]
            atr = df['volatility'].iloc[-1] * price
            risk_per_lot = atr * 1.5  # 1.5 * ATR как стоп

            if prob > 0.55:
                signals.append({
                    'figi': figi,
                    'price': price,
                    'prob': prob,
                    'risk': risk_per_lot,
                    'tp': price * 1.02,
                    'sl': price * 0.985
                })
        except Exception as e:
            await send_telegram_message(f"Ошибка анализа {figi}: {str(e)}")
    return signals

async def execute_signals(client, signals):
    for sig in signals:
        try:
            if not risk_manager.can_add_position(sig['figi'], sig['risk']):
                continue

            # 1 лот
            quantity = 1
            order = await client.orders.post_order(
                figi=sig['figi'],
                quantity=quantity,
                direction=OrderDirection.ORDER_DIRECTION_BUY,
                order_type=OrderType.ORDER_TYPE_MARKET,
                account_id=ACCOUNT_ID
            )

            save_signal({
                'figi': sig['figi'],
                'price': sig['price'],
                'signal': 'BUY',
                'prob': sig['prob'],
                'position_size': quantity,
                'tp': sig['tp'],
                'sl': sig['sl']
            })
            save_operation({
                'id': order.order_id,
                'timestamp': datetime.now().isoformat(),
                'figi': sig['figi'],
                'operation_type': 'BUY',
                'price': sig['price'],
                'quantity': quantity,
                'status': 'executed'
            })

            risk_manager.commit_risk(sig['figi'], sig['risk'])
            await send_telegram_message(
                f"🟢 СИГНАЛ: {sig['figi']}\n"
                f"Цена: {sig['price']:.0f} | Вероятность: {sig['prob']:.2f}\n"
                f"SL: {sig['sl']:.0f}, TP: {sig['tp']:.0f}"
            )
        except Exception as e:
            await send_telegram_message(f"❌ Ошибка ордера {sig['figi']}: {str(e)}")

async def portfolio_monitor(client):
    try:
        portfolio = await client.operations.get_portfolio(account_id=ACCOUNT_ID)
        for pos in portfolio.positions:
            figi = pos.figi
            quantity = pos.quantity.units
            avg_price = quote_to_float(pos.average_position_price)
            current_price = quote_to_float(pos.current_price) if pos.current_price else avg_price
            name = "Unknown"
            try:
                instr = await client.instruments.get_instrument_by(id_type=2, id=figi)
                name = instr.instrument.name
            except:
                pass
            update_portfolio(figi, name, quantity, avg_price, current_price)
    except Exception as e:
        await send_telegram_message(f"Ошибка обновления портфеля: {str(e)}")

async def run_trading_cycle():
    try:
        async with AsyncClient(TINKOFF_TOKEN) as client:
            risk_manager.update_balance()
            signals = await analyze_figis(client)
            if signals:
                await execute_signals(client, signals)
            await portfolio_monitor(client)
    except Exception as e:
        await send_telegram_message(f"Критическая ошибка цикла: {str(e)}")
